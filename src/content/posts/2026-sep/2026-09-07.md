class Solution {
public:
  double findMedianSortedArrays(vector<int> &nums1, vector<int> &nums2) {

    int target = nums1.size() + nums2.size();
    int mid = target >> 1;

    auto n1 = nums1.begin(), n2 = nums2.begin();
    int cur, prv;

    while (n1 != nums1.end() && n2 != nums2.end()) {

      prv = cur;

      if (*n1 < *n2) {
        cur = *n1;
        n1++;
      } else {
        cur = *n2;
        n2++;
      }

      --mid;

      if (mid < 0) {
        break;
      }
    }

    while (n1 != nums1.end() && mid >= 0) {
      prv = cur;
      cur = *n1;

      n1++;
      mid--;
    }

    while (n2 != nums2.end() && mid >= 0) {
      prv = cur;
      cur = *n2;

      n2++;
      mid--;
    }

    if (target % 2) {
      return cur;
    } else {
      return (double)(prv + cur) / 2;
    }
  }
};
